package com.fulfilment.application.monolith.warehouses.domain.usecases;

import com.fulfilment.application.monolith.warehouses.domain.models.Warehouse;
import com.fulfilment.application.monolith.warehouses.domain.models.Location;
import com.fulfilment.application.monolith.warehouses.domain.ports.CreateWarehouseOperation;
import com.fulfilment.application.monolith.warehouses.domain.ports.LocationResolver;
import com.fulfilment.application.monolith.warehouses.domain.ports.WarehouseStore;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.ws.rs.WebApplicationException;
import jakarta.ws.rs.core.Response;
import java.util.List;

@ApplicationScoped
public class CreateWarehouseUseCase implements CreateWarehouseOperation {

  private final WarehouseStore warehouseStore;
  private final LocationResolver locationResolver;

  public CreateWarehouseUseCase(WarehouseStore warehouseStore, LocationResolver locationResolver) {
    this.warehouseStore = warehouseStore;
    this.locationResolver = locationResolver;
  }

  @Override
  public void create(Warehouse warehouse) {

    // 1. Resolve Location limits
    Location cityLimit = locationResolver.resolveByIdentifier(warehouse.location);

    if (cityLimit == null) {
      throw new WebApplicationException("The provided location is not supported", Response.Status.BAD_REQUEST);
    }

    // 2. Fetch existing warehouses in this city to check global limits
    List<Warehouse> existingWarehouses = warehouseStore.getAll().stream()
        .filter(w -> w.location.equalsIgnoreCase(warehouse.location))
        .filter(w -> w.archivedAt == null) // Only count active ones
        .toList();

    // 3. Check Max Number of Warehouses Rule
    if (existingWarehouses.size() >= cityLimit.maxNumberOfWarehouses) {
      throw new WebApplicationException(
          "Maximum number of warehouses reached for " + warehouse.location, 
          Response.Status.BAD_REQUEST);
    }

    // 4. Check Total Capacity Rule
    int currentTotalCapacity = existingWarehouses.stream().mapToInt(w -> w.capacity).sum();
    if ((currentTotalCapacity + warehouse.capacity) > cityLimit.maxCapacity) {
      throw new WebApplicationException(
          "New warehouse exceeds the total allowed capacity for " + warehouse.location, 
          Response.Status.BAD_REQUEST);
    }

    // 5. Basic Validation
    if (warehouse.capacity == null || warehouse.capacity <= 0) {
      throw new WebApplicationException("Capacity must be greater than zero", Response.Status.BAD_REQUEST);
    }

    // 6. Default stock
    if (warehouse.stock == null) {
      warehouse.stock = 0;
    }

    // 7. Save
    warehouseStore.create(warehouse);
  }
}
