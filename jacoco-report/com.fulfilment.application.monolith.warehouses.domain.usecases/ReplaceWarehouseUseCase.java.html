package com.fulfilment.application.monolith.warehouses.domain.usecases;

import com.fulfilment.application.monolith.warehouses.domain.models.Location;
import com.fulfilment.application.monolith.warehouses.domain.models.Warehouse;
import com.fulfilment.application.monolith.warehouses.domain.ports.LocationResolver;
import com.fulfilment.application.monolith.warehouses.domain.ports.ReplaceWarehouseOperation;
import com.fulfilment.application.monolith.warehouses.domain.ports.WarehouseStore;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.ws.rs.WebApplicationException;
import jakarta.ws.rs.core.Response;
import java.util.List;

@ApplicationScoped
public class ReplaceWarehouseUseCase implements ReplaceWarehouseOperation {

  private final WarehouseStore warehouseStore;
  private final LocationResolver locationResolver;

  public ReplaceWarehouseUseCase(WarehouseStore warehouseStore, LocationResolver locationResolver) {
    this.warehouseStore = warehouseStore;
    this.locationResolver = locationResolver;
  }

  @Override
  public void replace(Warehouse newWarehouse) {

    // 1. Resolve Location limits
    Location cityLimit = locationResolver.resolveByIdentifier(newWarehouse.location);

    if (cityLimit == null) {
      throw new WebApplicationException("The provided location is not supported", Response.Status.BAD_REQUEST);
    }

    // 2. Business Rule: Check total capacity in the city
    // Filter out the warehouse being replaced (using businessUnitCode) so we don't double-count its capacity
    List<Warehouse> otherWarehousesInCity = warehouseStore.getAll().stream()
        .filter(w -> w.location.equalsIgnoreCase(newWarehouse.location))
        .filter(w -> w.archivedAt == null)
        .filter(w -> !w.businessUnitCode.equals(newWarehouse.businessUnitCode))
        .toList();

    int currentTotalCapacity = otherWarehousesInCity.stream().mapToInt(w -> w.capacity).sum();

    if ((currentTotalCapacity + newWarehouse.capacity) > cityLimit.maxCapacity) {
      throw new WebApplicationException(
          "Replacement exceeds the total allowed capacity for " + newWarehouse.location, 
          Response.Status.BAD_REQUEST);
    }

    // 3. Basic Validation: Capacity must be valid
    if (newWarehouse.capacity == null || newWarehouse.capacity <= 0) {
      throw new WebApplicationException("Capacity must be greater than zero", Response.Status.BAD_REQUEST);
    }
    
    // 4. Stock safety check: New capacity cannot be less than current stock
    if (newWarehouse.stock != null && newWarehouse.capacity < newWarehouse.stock) {
      throw new WebApplicationException("New capacity cannot be lower than current stock levels", Response.Status.CONFLICT);
    }

    // 5. Persistence: Update the warehouse
    warehouseStore.update(newWarehouse);
  }
}
