package com.fulfilment.application.monolith.warehouses.adapters.database;

import com.fulfilment.application.monolith.warehouses.domain.models.Warehouse;
import com.fulfilment.application.monolith.warehouses.domain.ports.WarehouseStore;
import io.quarkus.hibernate.orm.panache.PanacheRepository;
import jakarta.enterprise.context.ApplicationScoped;
import jakarta.transaction.Transactional;
import java.util.List;

@ApplicationScoped
public class WarehouseRepository implements WarehouseStore, PanacheRepository<DbWarehouse> {

  @Override
  public List<Warehouse> getAll() {
    // Converts all database records found into Domain objects for the Use Cases
    return this.listAll().stream()
        .map(DbWarehouse::toWarehouse)
        .toList();
  }

  @Override
  @Transactional // Required for writing to DB
  public void create(Warehouse warehouse) {
    // Convert domain model to entity and persist
    DbWarehouse entity = DbWarehouse.fromWarehouse(warehouse);
    this.persist(entity);
  }

  @Override
  @Transactional // Required for updating DB
  public void update(Warehouse warehouse) {
    // 1. Find the existing entity in the DB using the unique business code
    DbWarehouse existing = find("businessUnitCode", warehouse.businessUnitCode).firstResult();
    
    if (existing != null) {
      // 2. Update the fields. Hibernate will automatically save these changes 
      // at the end of the method because of @Transactional.
      existing.location = warehouse.location;
      existing.capacity = warehouse.capacity;
      existing.stock = warehouse.stock;
      existing.archivedAt = warehouse.archivedAt;
    }
  }

  @Override
  @Transactional // Required for deleting from DB
  public void remove(Warehouse warehouse) {
    // Delete the record matching the business unit code
    delete("businessUnitCode", warehouse.businessUnitCode);
  }

  @Override
  public Warehouse findByBusinessUnitCode(String buCode) {
    // Search for a single warehouse by its code
    DbWarehouse entity = find("businessUnitCode", buCode).firstResult();
    
    // Return null if not found, or the Domain model if it exists
    return entity != null ? entity.toWarehouse() : null;
  }
}
